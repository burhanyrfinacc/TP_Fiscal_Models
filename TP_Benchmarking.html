import pandas as pd import numpy as np from openpyxl import Workbook from openpyxl.styles import Font, Alignment from openpyxl.chart import BarChart, Reference from openpyxl.utils.dataframe import dataframe_to_rows from openpyxl.utils import get_column_letter from openpyxl.worksheet.worksheet import MergedCell import warnings warnings.filterwarnings('ignore') class TPBenchmarkingModel: def __init__(self): self.comparables = None self.tested_party = None self.method = 'TNMM' def load_netflix_comparables(self): """Load Netflix (NFLX) + 10 real entertainment comparables""" data = { 'Company': [ 'Netflix Inc. (NFLX)', # Tested Party 'Spotify Technology S.A. (SPOT)', 'Warner Bros. Discovery (WBD)', 'The Walt Disney Company (DIS)', 'Paramount Global (PARA)', 'Tencent Music Entertainment (TME)', 'Live Nation Entertainment (LYV)', 'Roblox Corporation (RBLX)', 'Electronic Arts Inc. (EA)', 'NetEase, Inc. (NTES)', 'Comcast Corporation (CMCSA)' ], 'Revenue': [33610, 15170, 40294, 88933, 29730, 3380, 18900, 4560, 7410, 15480, 121572], # $M 'COGS': [19400, 11600, 32000, 61000, 21000, 2400, 13500, 3200, 4200, 9800, 75300], 'Operating_Expenses': [9500, 2900, 6500, 21000, 6500, 800, 4600, 1100, 2100, 4200, 32500], 'Operating_Profit': [3710, 670, 1794, 5853, 2230, 180, 800, 260, 1110, 1480, 12772] } self.comparables = pd.DataFrame(data) # Calculate PLIs (%) self.comparables['Gross_Margin'] = ((self.comparables['Revenue'] - self.comparables['COGS']) / self.comparables['Revenue'] * 100) self.comparables['Operating_Margin'] = (self.comparables['Operating_Profit'] / self.comparables['Revenue'] * 100) self.comparables['Cost_Plus_Markup'] = (self.comparables['Operating_Profit'] / (self.comparables['COGS'] + self.comparables['Operating_Expenses']) * 100) print("Loaded Netflix + 10 comparables") print(self.comparables[['Company', 'Operating_Margin']].round(2)) def set_tested_party(self): """Netflix as tested party - extracted from comparables[0]""" netflix_row = self.comparables.iloc[0] self.tested_party = pd.DataFrame({ 'Entity': ['Netflix Inc. (NFLX) - TESTED PARTY'], 'Revenue': [netflix_row['Revenue']], 'COGS': [netflix_row['COGS']], 'Operating_Expenses': [netflix_row['Operating_Expenses']], 'Operating_Profit': [netflix_row['Operating_Profit']] }) self.tested_party['Operating_Margin'] = [netflix_row['Operating_Margin']] self.tested_party['Gross_Margin'] = [netflix_row['Gross_Margin']] self.tested_party['Cost_Plus_Markup'] = [netflix_row['Cost_Plus_Markup']] print(f" Netflix set as Tested Party: OM = {self.tested_party['Operating_Margin'].iloc[0]:.2f}%") def calculate_iqr_range(self, data): q1 = data.quantile(0.35) # 35th-75th percentile q3 = data.quantile(0.75) return { 'Q1_35th': q1, 'Median': data.median(), 'Q3_75th': q3 } def run_benchmarking_analysis(self): if self.comparables is None or self.tested_party is None: raise ValueError("Run load_netflix_comparables() and set_tested_party() first!") # Use comparables[1:] to exclude Netflix (tested party) comp_data = self.comparables.iloc[1:][self.method_metrics[self.method]] iqr = self.calculate_iqr_range(comp_data) tested_pli = self.tested_party[self.method_metrics[self.method]].iloc[0] compliant = iqr['Q1_35th'] <= tested_pli <= iqr['Q3_75th'] results = { 'Method': self.method, 'Tested_Party_PLI': f"{tested_pli:.2f}%", 'Arms_Length_Range': f"{iqr['Q1_35th']:.2f}% - {iqr['Q3_75th']:.2f}%", 'Median': f"{iqr['Median']:.2f}%", 'Compliant': 'YES' if compliant else 'NO', 'n_Comparables': len(comp_data) } print("NETFLIX TNMM BENCHMARKING RESULTS") print("=" * 60) for k, v in results.items(): print(f"{k:25}: {v}") return results, iqr def generate_excel_report(self, filename="Netflix_TP_Analysis.xlsx"): wb = Workbook() ws = wb.active ws.title = "Netflix TNMM Analysis" # Title ws['A1'] = "NETFLIX TRANSFER PRICING BENCHMARKING (TNMM)" ws.merge_cells('A1:K1') ws['A1'].font = Font(size=16, bold=True) ws['A1'].alignment = Alignment(horizontal='center') # Summary Results row = 4 ws[f'A{row}'] = "TESTED PARTY: NETFLIX INC. (NFLX)"; row += 2 ws[f'A{row}'] = "Method"; ws[f'B{row}'] = self.method; row += 1 ws[f'A{row}'] = "Netflix Operating Margin"; ws[f'B{row}'] = f"{self.tested_party['Operating_Margin'].iloc[0]:.2f}%"; row += 1 # Recalculate IQR for display comp_data = self.comparables.iloc[1:]['Operating_Margin'] iqr = self.calculate_iqr_range(comp_data) ws[f'A{row}'] = "Arm's Length Range (35th-75th)"; ws[f'B{row}'] = f"{iqr['Q1_35th']:.2f}% - {iqr['Q3_75th']:.2f}%"; row += 2 # Compliance status compliant = iqr['Q1_35th'] <= self.tested_party['Operating_Margin'].iloc[0] <= iqr['Q3_75th'] ws[f'A{row}'] = "COMPLIANT"; ws[f'B{row}'] = 'YES' if compliant else 'NO'; row += 3 # Comparables table (exclude tested party) ws[f'A{row}'] = "COMPARABLE COMPANIES (10 peers)"; row += 2 df_export = self.comparables.iloc[1:][['Company', 'Revenue', 'Operating_Margin', 'Gross_Margin', 'Cost_Plus_Markup']].round(2) df_export.insert(0, 'Rank', range(1, len(df_export)+1)) start_row = row + 1 for r in dataframe_to_rows(df_export, index=False, header=True): ws.append(r) # Auto-size columns (handles merged cells) for col_num in range(1, ws.max_column + 1): col_letter = get_column_letter(col_num) max_length = 0 for cell in ws[col_letter]: if isinstance(cell, MergedCell): continue if cell.value: max_length = max(max_length, len(str(cell.value))) ws.column_dimensions[col_letter].width = min(max_length + 2, 25) wb.save(filename) print(f"\nExcel report saved: {filename}") return filename method_metrics = {'TNMM': 'Operating_Margin', 'RPM': 'Gross_Margin', 'CPM': 'Cost_Plus_Markup'} # RUN NETFLIX ANALYSIS model = TPBenchmarkingModel() model.load_netflix_comparables() model.set_tested_party() results, iqr = model.run_benchmarking_analysis() report = model.generate_excel_report("Netflix_vs_10_Comps_TNMM.xlsx") print("Netflix TP Analysis Complete") Loaded Netflix + 10 comparables Company Operating_Margin 0 Netflix Inc.
